1. OBJECTIVE & CODE REFACTORING DETAILS:
--------------------------------------------------------------------------------
In Exercise 1, every process allocated the full 1,000,000-element array (4 MB each)
and received the entire array via MPI_Bcast. This is highly inefficient in terms of
both memory consumption and network bandwidth, as each process only needs its own
chunk of size N / size (250,000 elements for 4 processes = 1 MB).

Key modifications made in sum_scatter.c:
1. Conditional Allocation on Root:
   Only rank 0 allocates and initializes the full 1,000,000-element array (`array`).
   Non-root processes set `array = NULL`, saving (P - 1) * 4 MB of RAM.
2. Local Chunk Buffer:
   Every process allocates a buffer `local_chunk` of size `chunk_size = N / size`.
3. MPI_Scatter Integration:
   Replaced MPI_Bcast with:
   MPI_Scatter(array, chunk_size, MPI_INT, local_chunk, chunk_size, MPI_INT, 0, MPI_COMM_WORLD);
   Note: The `sendcount` argument in MPI_Scatter specifies the number of elements
   sent to EACH process (chunk_size), NOT the total size N.
4. Local Computation Loop:
   Loop index runs over local_chunk[0 .. chunk_size - 1] instead of indexing into the
   global array.
5. Collection via Send/Recv:
   Point-to-point communication (MPI_Send and MPI_Recv) is retained as required.

2. SOURCE CODE (sum_scatter.c):
--------------------------------------------------------------------------------
#include <mpi.h>
#include <stdio.h>
#include <stdlib.h>

#define N 1000000

int main(int argc, char **argv) {
    MPI_Init(&argc, &argv);
    int rank, size;
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    int chunk_size = N / size;

    /* Only root allocates the full array */
    int *array = NULL;
    if (rank == 0) {
        array = (int *)malloc(N * sizeof(int));
        if (array == NULL) {
            fprintf(stderr, "Root failed to allocate full array\n");
            MPI_Abort(MPI_COMM_WORLD, 1);
        }
        for (int i = 0; i < N; i++)
            array[i] = i + 1;
        printf("Root filled array with values 1 to %d\n", N);
    }

    /* All processes allocate only their small local chunk */
    int *local_chunk = (int *)malloc(chunk_size * sizeof(int));
    if (local_chunk == NULL) {
        fprintf(stderr, "Rank %d failed to allocate local_chunk\n", rank);
        MPI_Abort(MPI_COMM_WORLD, 1);
    }

    double start = MPI_Wtime();

    /*
     * SCATTER: Root divides array and sends chunk_size elements to each process.
     * sendcount is chunk_size (number of elements to each process, NOT total N).
     */
    MPI_Scatter(array, chunk_size, MPI_INT,
                local_chunk, chunk_size, MPI_INT,
                0, MPI_COMM_WORLD);

    /*
     * Each process computes the sum of its local chunk [0 .. chunk_size - 1].
     */
    long long local_sum = 0;
    for (int i = 0; i < chunk_size; i++)
        local_sum += local_chunk[i];

    int start_idx = rank * chunk_size;
    int end_idx = start_idx + chunk_size;
    printf("  Rank %d: summed indices [%d, %d) => local_sum = %lld\n",
           rank, start_idx, end_idx, local_sum);

    /* Collect results using point-to-point communication (unchanged) */
    if (rank != 0) {
        MPI_Send(&local_sum, 1, MPI_LONG_LONG, 0, 0, MPI_COMM_WORLD);
    } else {
        long long total_sum = local_sum;
        for (int r = 1; r < size; r++) {
            long long recv_sum;
            MPI_Recv(&recv_sum, 1, MPI_LONG_LONG, r, 0,
                     MPI_COMM_WORLD, MPI_STATUS_IGNORE);
            total_sum += recv_sum;
        }
        double elapsed = MPI_Wtime() - start;
        long long expected = (long long)N * (N + 1) / 2;
        printf("\n[Scatter] Total sum   = %lld\n", total_sum);
        printf("[Scatter] Expected    = %lld\n", expected);
        printf("[Scatter] Correct?    = %s\n", total_sum == expected ? "YES" : "NO");
        printf("[Scatter] Time        = %.4f sec\n", elapsed);
    }

    if (rank == 0) free(array);
    free(local_chunk);
    MPI_Finalize();
    return 0;
}
